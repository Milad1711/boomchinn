     3→ * Three.js (render) + cannon-es (physics).
     4→ *
     5→ * Systems:
     6→ *  - Mergeable objects with 5 tiers; two same-tier objects merge on contact.
     7→ *  - Drag & drop touch input (kinematic hold).
     8→ *  - Guided-missile cannon: tap an obstacle and the turret fires a homing
     9→ *    rocket that ALWAYS hits it. Precise, zero wasted shots, clear role.
    10→ *  - Destructible obstacles (wood crate / glass crystal / stone) with
    11→ *    floating HP bars, splash damage and chain reactions.
    12→ *  - Timed auto-spawn waves keep levels long and never dead-end.
    13→ *  - Idle auto-hint: after a few seconds the game itself highlights the
    14→ *    two closest mergeable items.
    15→ *  - Combos, score, goal tracking, win/lose detection.
    16→ *  - Particles, debris, shockwaves, floaties, shake, sounds, haptics.
    17→ */

   100→function easeOutBack(t: number): number {
   101→  const c1 = 1.70158
   102→  const c3 = c1 + 1
   103→  return 1 + c3 * Math.pow(t - 1, 3) + c1 * Math.pow(t - 1, 2)
   104→}

   106→function roundedRectPath(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   107→  const rad = Math.min(r, h / 2, w / 2)
   108→  ctx.beginPath()
   109→  ctx.moveTo(x + rad, y)
   110→  ctx.arcTo(x + w, y, x + w, y + h, rad)
   111→  ctx.arcTo(x + w, y + h, x, y + h, rad)
   112→  ctx.arcTo(x, y + h, x, y, rad)
   113→  ctx.arcTo(x, y, x + w, y, rad)
   114→  ctx.closePath()
   115→}

   325→  private makeGradientTexture(): THREE.CanvasTexture {
   326→    const c = document.createElement('canvas')
   327→    c.width = 2
   328→    c.height = 512
   329→    const ctx = c.getContext('2d')!
   330→    const g = ctx.createLinearGradient(0, 0, 0, 512)
   331→    g.addColorStop(0, '#e6fcf3')
   332→    g.addColorStop(1, '#8ce6d2')
   333→    ctx.fillStyle = g
   334→    ctx.fillRect(0, 0, 2, 512)
   335→    const tex = new THREE.CanvasTexture(c)
   336→    tex.colorSpace = THREE.SRGBColorSpace
   337→    return tex

   340→  private makeFloorTexture(): THREE.CanvasTexture {
   341→    const c = document.createElement('canvas')
   342→    c.width = 256
   343→    c.height = 256
   344→    const ctx = c.getContext('2d')!
   345→    ctx.fillStyle = '#c3f2df'
   346→    ctx.fillRect(0, 0, 256, 256)

   348→    for (let y = 0; y < 4; y++)
   349→      for (let x = 0; x < 4; x++) if ((x + y) % 2 === 0) ctx.fillRect(x * 64, y * 64, 64, 64)

   352→    tex.wrapT = THREE.RepeatWrapping
   353→    tex.repeat.set(11, 11)
   354→    tex.colorSpace = THREE.SRGBColorSpace
   355→    tex.anisotropy = 4
   356→    return tex

   359→  /** Warm wooden planks texture for crates. */
   360→  private makeWoodTexture(): THREE.CanvasTexture {
   361→    const c = document.createElement('canvas')
   362→    c.width = 128
   363→    c.height = 128
   364→    const ctx = c.getContext('2d')!
   365→    const shades = ['#cf9058', '#c68a52', '#d29a63', '#c2854f']
   366→    for (let i = 0; i < 4; i++) {
   367→      ctx.fillStyle = shades[i]
   368→      ctx.fillRect(0, i * 32, 128, 32)
   369→      // grain streaks
   370→      ctx.strokeStyle = 'rgba(122,74,38,0.28)'
   371→      ctx.lineWidth = 1.4
   372→      for (let g = 0; g < 3; g++) {
   373→        const gy = i * 32 + 6 + g * 9 + Math.random() * 4
   374→        ctx.beginPath()
   375→        ctx.moveTo(4, gy)
   376→        ctx.bezierCurveTo(40, gy - 3, 80, gy + 3, 124, gy)
   377→        ctx.stroke()
   378→      }
   379→      // plank separator
   380→      ctx.fillStyle = 'rgba(96,58,28,0.75)'
   381→      ctx.fillRect(0, i * 32, 128, 2.4)
   382→    }
   383→    // nails
   384→    ctx.fillStyle = '#7a4a26'
   385→    for (const [nx, ny] of [[8, 6], [120, 6], [8, 122], [120, 122]]) {
   386→      ctx.beginPath()
   387→      ctx.arc(nx, ny, 2.6, 0, Math.PI * 2)
   388→      ctx.fill()
   389→    }
   390→    const tex = new THREE.CanvasTexture(c)
   391→    tex.colorSpace = THREE.SRGBColorSpace
   392→    tex.anisotropy = 4
   393→    return tex
   394→  }

   396→  /** Speckled granite texture for stones. */
   397→  private makeStoneTexture(): THREE.CanvasTexture {
   398→    const c = document.createElement('canvas')
   399→    c.width = 128
   400→    c.height = 128
   401→    const ctx = c.getContext('2d')!
   402→    ctx.fillStyle = '#a7b3ad'
   403→    ctx.fillRect(0, 0, 128, 128)
   404→    for (let i = 0; i < 260; i++) {
   405→      const shade = Math.random()
   406→      ctx.fillStyle =
   407→        shade > 0.6 ? 'rgba(120,132,126,0.5)' : shade > 0.25 ? 'rgba(178,190,184,0.5)' : 'rgba(88,100,94,0.4)'
   408→      ctx.beginPath()
   409→      ctx.arc(Math.random() * 128, Math.random() * 128, 1 + Math.random() * 3.4, 0, Math.PI * 2)
   410→      ctx.fill()
   411→    }
   412→    ctx.strokeStyle = 'rgba(84,96,90,0.5)'
   413→    ctx.lineWidth = 1.6
   414→    for (let i = 0; i < 4; i++) {
   415→      ctx.beginPath()
   416→      const sx = Math.random() * 128
   417→      const sy = Math.random() * 128
   418→      ctx.moveTo(sx, sy)
   419→      ctx.lineTo(sx + (Math.random() - 0.5) * 60, sy + (Math.random() - 0.5) * 60)
   420→      ctx.stroke()
   421→    }
   422→    const tex = new THREE.CanvasTexture(c)
   423→    tex.colorSpace = THREE.SRGBColorSpace
   424→    tex.anisotropy = 4
   425→    return tex
   426→  }

   428→  private buildEnvironment() {
   429→    const { trayW: W, trayD: D } = this.def
   430→    const t = 0.5
   431→    const wallH = 1.7
   432→    const frontH = 0.85
   433→
   434→    // outer ground
   435→    const ground = new THREE.Mesh(
   436→      new THREE.PlaneGeometry(60, 60),
   437→      new THREE.MeshStandardMaterial({ map: this.makeFloorTexture(), roughness: 0.85, metalness: 0 })
   438→    )
   439→    ground.rotation.x = -Math.PI / 2
   440→    ground.position.y = -0.01
   441→    ground.receiveShadow = true
   442→    this.scene.add(ground)
   443→
   444→    // tray inner floor
   445→    const tray = new THREE.Mesh(
   446→      new THREE.PlaneGeometry(W, D),
   447→      new THREE.MeshStandardMaterial({ color: 0xeefcf5, roughness: 0.5, metalness: 0.02 })
   448→    )
   449→    tray.rotation.x = -Math.PI / 2
   450→    tray.position.y = 0.005
   451→    tray.receiveShadow = true
   452→    this.scene.add(tray)
   453→
   454→    // walls
   455→    const wallMat = new THREE.MeshStandardMaterial({ color: 0x77dbc3, roughness: 0.35, metalness: 0.05 })
   456→    const addWall = (w: number, h: number, d: number, x: number, y: number, z: number) => {
   457→      const mesh = new THREE.Mesh(new RoundedBoxGeometry(w, h, d, 2, 0.1), wallMat)
   458→      mesh.position.set(x, y, z)
   459→      mesh.castShadow = true
   460→      mesh.receiveShadow = true
   461→      this.scene.add(mesh)
   462→      const body = new CANNON.Body({
   463→        mass: 0,
   464→        shape: new CANNON.Box(new CANNON.Vec3(w / 2, h / 2, d / 2)),
   465→        material: this.matGround,
   466→      })
   467→      body.position.set(x, y, z)
   468→      this.world.addBody(body)
   469→    }
   470→    addWall(W + 2 * t, wallH, t, 0, wallH / 2, -D / 2 - t / 2) // back
   471→    addWall(W + 2 * t, frontH, t, 0, frontH / 2, D / 2 + t / 2) // front (low)
   472→    addWall(t, wallH, D, -W / 2 - t / 2, wallH / 2, 0) // left
   473→    addWall(t, wallH, D, W / 2 + t / 2, wallH / 2, 0) // right
   474→  }

   476→  private buildCannon() {
   477→    // no obstacles -> no cannon needed (missiles only target obstacles)

   481→    this.cannonRangeX = W / 2 + 0.7
   482→    this.cannonGroup.position.copy(this.cannonPos)

   485→    // plinth base (dark teal)

   488→      new THREE.MeshStandardMaterial({ color: 0x1e5c4e, roughness: 0.45, metalness: 0.3 })
   489→    )
   490→    base.position.y = 0.25
   491→    base.castShadow = true
   492→    this.cannonGroup.add(base)
   493→
   494→    // amber accent ring
   495→    const ring = new THREE.Mesh(
   496→      new THREE.CylinderGeometry(0.58, 0.58, 0.1, 24),
   497→      new THREE.MeshStandardMaterial({ color: 0xf59e0b, roughness: 0.35, metalness: 0.4 })
   498→    )
   499→    ring.position.y = 0.53

   502→    // rotating turret + barrel (recoils on fire)
   503→    const pivot = new THREE.Group()
   504→    pivot.position.y = 0.72
   505→    pivot.rotation.x = -AIM_ELEVATION
   506→    this.barrelGroup.position.z = 0

   511→    barrel.rotation.x = -Math.PI / 2
   512→    barrel.position.z = -0.78
   513→    barrel.castShadow = true

   524→    this.scene.add(this.cannonGroup)
   525→  }

   527→  /** Pulsing reticle shown on the turret's current auto-target. */
   528→  private buildReticle() {
   529→    if (this.def.obstacles.length === 0) return
   530→    this.reticleMat = new THREE.MeshBasicMaterial({
   531→      color: 0xf59e0b,
   532→      transparent: true,
   533→      opacity: 0.9,
   534→      depthWrite: false,
   535→      side: THREE.DoubleSide,
   536→    })

   768→  private onPointerDown = (e: PointerEvent) => {
   769→    audio.unlock()
   770→    if (this.paused || this.won || this.lost || this.disposed) return
   771→    this.setNdc(e)
   772→    this.lastActionTime = this.time
   773→    this.hintTextShown = false
   774→
   775→    // 1) pick a mergeable object
   776→    this.raycaster.setFromCamera(this.ndc, this.camera)
   777→    const meshes: THREE.Mesh[] = []
   778→    for (const it of this.items.values()) if (!it.consumed) meshes.push(it.mesh)
   779→    const hits = this.raycaster.intersectObjects(meshes, false)
   780→    if (hits.length > 0) {
   781→      const item = hits[0].object.userData.item as MergeItem
   782→      this.held = item
   783→      item.held = true
   784→      item.body.type = CANNON.Body.KINEMATIC
   785→      item.body.velocity.setZero()
   786→      item.body.angularVelocity.scale(0.2, item.body.angularVelocity)
   787→      item.body.wakeUp()
   788→      this.holdPlane.set(new THREE.Vector3(0, 1, 0), -HOLD_Y)
   789→      this.updateDragPoint()
   790→      audio.pick()
   791→      this.canvas.setPointerCapture(e.pointerId)
   792→      return
   793→    }
   794→
   795→    // 2) tap an obstacle -> fire a guided missile at it (always hits)
   796→    const obMeshes = this.obstacles.map((o) => o.mesh)
   797→    const obHits = this.raycaster.intersectObjects(obMeshes, false)
   798→    if (obHits.length > 0) {
   799→      const obstacle = this.obstacles.find((o) => o.mesh === obHits[0].object)
   800→      if (obstacle) this.fireMissileAt(obstacle)
   801→    }
   802→  }

   852→  /** Fire a guided missile from the turret at the tapped obstacle. */
   853→  private fireMissileAt(target: ObstacleItem) {
   854→    if (!this.cannonReady || this.shots <= 0 || this.won || this.lost || target.dead) return
   855→    this.shots--
   856→    this.shotsFired++
   857→    this.lastShotTime = this.time
   858→    this.lastActionTime = this.time
   859→    this.hintTextShown = false
   860→    this.cb.onShots(this.shots)
   861→
   862→    let bomb = false
   863→    if (this.bombArmed) {
   864→      bomb = true
   865→      this.bombArmed = false
   866→      this.cb.onBombArmed(false)
   867→      this.cb.onBombConsumed()
   868→    }
   869→
   870→    // snap the turret toward the target, then launch from the muzzle
   871→    const dx = target.body.position.x - this.cannonGroup.position.x
   872→    const dz = target.body.position.z - this.cannonPos.z
   873→    this.aimAngle = THREE.MathUtils.clamp(Math.atan2(dx, -dz), -MAX_AIM_ANGLE, MAX_AIM_ANGLE)
   874→
   875→    const dir = new THREE.Vector3(Math.sin(this.aimAngle), 0, -Math.cos(this.aimAngle))
   876→    const from = new THREE.Vector3(
   877→      this.cannonGroup.position.x + dir.x * 1.1,
   878→      MISSILE_Y,
   879→      this.cannonPos.z + dir.z * 1.1
   880→    )
   881→    const to = new THREE.Vector3(target.body.position.x, target.body.position.y, target.body.position.z)
   882→    const dist = from.distanceTo(to)
   883→    const ctrl = from.clone().add(to).multiplyScalar(0.5)
   884→    ctrl.y += Math.min(2.2, 0.9 + dist * 0.22)
   885→
   886→    audio.shoot()
   887→    this.buzz(25)
   888→    this.shake = Math.max(this.shake, 0.14)
   889→    this.recoil = 1
   890→    this.fx.burst(from, '#ffd27a', 10, 3.4, 1.4)
   891→
   892→    const skin = BALL_SKINS.find((s) => s.id === this.skinId) ?? BALL_SKINS[0]
   893→    this.lastBallColor = skin.color
   894→    const mat = new THREE.MeshStandardMaterial({
   895→      color: bomb ? '#ffb14d' : skin.color,
   896→      roughness: 0.25,
   897→      metalness: 0.4,
   898→      emissive: new THREE.Color(bomb ? '#ff7a1f' : skin.color),
   899→      emissiveIntensity: bomb ? 0.8 : 0.35,
   900→    })
   901→    const geo = new THREE.CapsuleGeometry(0.13, 0.3, 4, 10)
   902→    geo.rotateX(Math.PI / 2) // length along Z so lookAt points the nose forward
   903→    const mesh = new THREE.Mesh(geo, mat)
   904→    mesh.position.copy(from)
   905→    this.scene.add(mesh)
   906→
   907→    this.missiles.push({
   908→      mesh,
   909→      from,
   910→      ctrl,
   911→      target,
   912→      t: 0,
   913→      dur: THREE.MathUtils.clamp(0.38 + dist * 0.05, 0.5, 0.95),
   914→      bomb,
   915→      trailTimer: 0,
   916→    })
   917→    this.cb.onBallActive(true)
   918→  }

   920→  /** Bezier flight + live retargeting + impact. */
   921→  private updateMissiles(dt: number) {
   922→    for (const m of [...this.missiles]) {
   923→      m.t += dt / m.dur
   924→
   925→      // target vanished (destroyed by something else) -> steer to a new one
   926→      if (m.target?.dead) {
   927→        m.target = this.nearestObstacle(m.mesh.position)
   928→        if (m.target) {
   929→          m.from.copy(m.mesh.position)
   930→          const to = new THREE.Vector3(m.target.body.position.x, m.target.body.position.y, m.target.body.position.z)
   931→          const dist = m.from.distanceTo(to)
   932→          m.ctrl = m.from.clone().add(to).multiplyScalar(0.5)
   933→          m.ctrl.y += Math.min(1.8, 0.6 + dist * 0.2)
   934→          m.t = 0
   935→          m.dur = THREE.MathUtils.clamp(0.3 + dist * 0.05, 0.35, 0.7)
   936→        }
   937→      }
   938→
   939→      const end =
   940→        m.target && !m.target.dead
   941→          ? new THREE.Vector3(m.target.body.position.x, m.target.body.position.y, m.target.body.position.z)
   942→          : m.mesh.position.clone()
   943→
   944→      const t = Math.min(m.t, 1)
   945→      const a = m.from.clone().lerp(m.ctrl, t)
   946→      const b = m.ctrl.clone().lerp(end, t)
   947→      const pos = a.lerp(b, t)
   948→      const vel = pos.clone().sub(m.mesh.position)
   949→      m.mesh.position.copy(pos)
   950→      if (vel.lengthSq() > 1e-6) m.mesh.lookAt(pos.clone().add(vel))
   951→
   952→      // exhaust trail
   953→      m.trailTimer += dt
   954→      if (m.trailTimer > 0.03) {
   955→        m.trailTimer = 0
   956→        this.fx.burst(pos, m.bomb ? '#ffb14d' : this.lastBallColor, 2, 0.6, 0.35)
   957→      }
   958→
   959→      if (m.t >= 1) {
   960→        this.missiles = this.missiles.filter((x) => x !== m)
   961→        this.scene.remove(m.mesh)
   962→        m.mesh.geometry.dispose()
   963→        ;(m.mesh.material as THREE.Material).dispose()
   964→        this.impactMissile(m)
   965→      }
   966→    }
   967→    if (this.missiles.length === 0) this.cb.onBallActive(false)
   968→  }

   970→  /** Damage, splash, physics impulse and FX exactly where the missile lands. */
   971→  private impactMissile(m: MissileItem) {
   972→    const p = new THREE.Vector3(m.mesh.position.x, Math.max(0.4, m.mesh.position.y), m.mesh.position.z)
   973→    const c = new CANNON.Vec3(p.x, p.y, p.z)
   974→    if (m.bomb) {
   975→      this.fx.burst(p, '#ffb14d', 40, 7, 3)
   976→      this.fx.ring(p, '#ffb14d', 3.6)
   977→      audio.blast()
   978→      this.shake = Math.max(this.shake, 0.7)
   979→      this.buzz(70)
   980→      this.explode(c, 3.4, 9)
   981→      this.splashDamage(c, 3.2, BOMB_SPLASH_DMG)
   982→      if (m.target && !m.target.dead) this.damageObstacle(m.target, BOMB_DMG, c)
   983→      return
   984→    }
   985→    this.fx.burst(p, '#ffffff', 8, 2.4, 1)
   986→    this.shake = Math.max(this.shake, 0.18)
   987→    this.explode(c, 2.4, 4)
   988→    this.splashDamage(c, 1.5, 6)
   989→    if (m.target && !m.target.dead) this.damageObstacle(m.target, MISSILE_DMG[m.target.kind], c)
   990→    else audio.thud()
   991→  }

   993→  private nearestObstacle(from: THREE.Vector3): ObstacleItem | null {
   994→    let best: ObstacleItem | null = null
   995→    let bestD = Infinity
   996→    for (const o of this.obstacles) {
   997→      if (o.dead) continue
   998→      const d = Math.hypot(o.body.position.x - from.x, o.body.position.z - from.z)
   999→      if (d < bestD) {
  1000→        bestD = d
  1001→        best = o
  1002→      }
  1003→    }
  1004→    return best
  1005→  }

  1007→  private damageObstacle(o: ObstacleItem, dmg: number, from: CANNON.Vec3) {
  1008→    if (o.dead || this.won) return
  1009→    o.hp -= dmg
  1010→    o.flash = 1
  1011→    this.drawHpBar(o)
  1012→    audio.crack()
  1013→    this.fx.burst(new THREE.Vector3(from.x, Math.max(0.3, from.y), from.z), OBSTACLE_STYLE[o.kind].color, 8, 3, 1)
  1014→    if (o.hp <= 0) this.destroyObstacle(o)
  1015→  }

  1426→  /** Two closest same-tier items — used by the idle auto-hint. */
  1427→  private closestMergePair(): [MergeItem, MergeItem] | null {
  1428→    let best: [MergeItem, MergeItem] | null = null
  1429→    let bestD = Infinity
  1430→    const arr = [...this.items.values()].filter((i) => !i.consumed)
  1431→    for (let i = 0; i < arr.length; i++) {
  1432→      for (let j = i + 1; j < arr.length; j++) {
  1433→        if (arr[i].tier !== arr[j].tier) continue
  1434→        const d = arr[i].body.position.distanceTo(arr[j].body.position)
  1435→        if (d < bestD) {
  1436→          bestD = d
  1437→          best = [arr[i], arr[j]]
  1438→        }
  1439→      }
  1440→    }
  1441→    return best
  1442→  }
