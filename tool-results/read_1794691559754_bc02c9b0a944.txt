     1→/**
     2→ * STACK & SMASH - 3D game engine.
     3→ * Three.js (render) + cannon-es (physics).
     4→ *
     5→ * Systems:
     6→ *  - Mergeable objects with 5 tiers; two same-tier objects merge on contact.
     7→ *  - Drag & drop touch input (kinematic hold).
     8→ *  - Mechanical cannon: aim in the bottom strip, release to fire a ball.
     9→ *  - Destructible obstacles with HP, splash damage and chain reactions.
    10→ *  - Combos, score, goal tracking, win/lose detection.
    11→ *  - Particles, debris, shockwaves, floaties, shake, sounds, haptics.
    12→ */

    79→function easeOutBack(t: number): number {
    80→  const c1 = 1.70158
    81→  const c3 = c1 + 1
    82→  return 1 + c3 * Math.pow(t - 1, 3) + c1 * Math.pow(t - 1, 2)
    83→}

   160→  constructor(
   161→    canvas: HTMLCanvasElement,
   162→    fxLayer: HTMLElement,
   163→    levelIndex: number,
   164→    cb: EngineCallbacks
   165→  ) {
   166→    this.canvas = canvas
   167→    this.levelIndex = levelIndex
   168→    this.def = LEVELS[levelIndex] ?? LEVELS[0]
   169→    this.cb = cb
   170→    this.shots = this.def.shots
   171→    this.settings = { sfx: true, music: true, haptics: true, shadows: true }
   172→
   173→    // ---------------------------------------------------------- renderer
   174→    this.renderer = new THREE.WebGLRenderer({ canvas, antialias: true, powerPreference: 'high-performance' })
   175→    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2))
   176→    this.renderer.shadowMap.enabled = true
   177→    this.renderer.shadowMap.type = THREE.PCFShadowMap
   178→
   179→    // ------------------------------------------------------------- scene
   180→    this.scene = new THREE.Scene()
   181→    this.scene.background = this.makeGradientTexture()
   182→    this.scene.fog = new THREE.Fog(0x9fe8d4, 34, 70)
   183→
   184→    const pmrem = new THREE.PMREMGenerator(this.renderer)
   185→    this.scene.environment = pmrem.fromScene(new RoomEnvironment(), 0.04).texture
   186→    this.scene.environmentIntensity = 0.5
   187→    pmrem.dispose()
   188→
   189→    this.camera = new THREE.PerspectiveCamera(44, 1, 0.1, 120)
   190→
   191→    const hemi = new THREE.HemisphereLight(0xf2fffa, 0x2b9a7c, 0.85)
   192→    this.scene.add(hemi)
   193→    const sun = new THREE.DirectionalLight(0xffffff, 1.7)
   194→    sun.position.set(7, 14, 6)
   195→    sun.castShadow = true
   196→    sun.shadow.mapSize.set(1024, 1024)
   197→    sun.shadow.camera.left = -11
   198→    sun.shadow.camera.right = 11
   199→    sun.shadow.camera.top = 11
   200→    sun.shadow.camera.bottom = -11
   201→    sun.shadow.camera.far = 40
   202→    sun.shadow.bias = -0.0004
   203→    this.scene.add(sun)
   204→
   205→    // ----------------------------------------------------------- physics
   206→    this.world = new CANNON.World({ gravity: new CANNON.Vec3(0, -16, 0) })
   207→    this.world.broadphase = new CANNON.SAPBroadphase(this.world)
   208→    this.world.allowSleep = true
   209→    this.world.solver.iterations = 10
   210→    this.world.defaultContactMaterial.friction = 0.35
   211→    this.world.addContactMaterial(new CANNON.ContactMaterial(this.matGround, this.matObj, { friction: 0.38, restitution: 0.12 }))
   212→    this.world.addContactMaterial(new CANNON.ContactMaterial(this.matObj, this.matObj, { friction: 0.32, restitution: 0.18 }))
   213→    this.world.addContactMaterial(new CANNON.ContactMaterial(this.matBall, this.matObj, { friction: 0.25, restitution: 0.5 }))
   214→    this.world.addContactMaterial(new CANNON.ContactMaterial(this.matBall, this.matGround, { friction: 0.3, restitution: 0.55 }))
   215→
   216→    const floorBody = new CANNON.Body({ mass: 0, shape: new CANNON.Plane(), material: this.matGround })
   217→    floorBody.quaternion.setFromEuler(-Math.PI / 2, 0, 0)
   218→    this.world.addBody(floorBody)
   219→
   220→    // ------------------------------------------------------------- world
   221→    this.buildEnvironment()
   222→    this.obstacleGeo = new RoundedBoxGeometry(1, 1, 1, 2, 0.09)
   223→    this.buildCannon()
   224→    for (const s of this.def.spawns) this.makeItem(s.tier, s.x, 1.7, s.z)
   225→    for (const o of this.def.obstacles) this.makeObstacle(o)
   226→
   227→    this.fx = new Effects(this.scene, this.camera, fxLayer)
   228→
   229→    // ------------------------------------------------------------ events
   230→    canvas.addEventListener('pointerdown', this.onPointerDown)
   231→    canvas.addEventListener('pointermove', this.onPointerMove)
   232→    canvas.addEventListener('pointerup', this.onPointerUp)
   233→    canvas.addEventListener('pointercancel', this.onPointerCancel)
   234→    window.addEventListener('resize', this.onResize)
   235→
   236→    const goalTarget =
   237→      this.def.goal.type === 'merge'
   238→        ? this.def.goal.count ?? 1
   239→        : this.def.goal.type === 'clear'
   240→          ? this.def.obstacles.length
   241→          : this.def.goal.points ?? 0
   242→    this.cb.onGoal(0, goalTarget)
   243→    this.cb.onShots(this.shots)
   244→
   245→    this.layout()
   246→    this.loop()
   247→  }

   251→  private makeGradientTexture(): THREE.CanvasTexture {
   252→    const c = document.createElement('canvas')
   253→    c.width = 2
   254→    c.height = 512
   255→    const ctx = c.getContext('2d')!
   256→    const g = ctx.createLinearGradient(0, 0, 0, 512)
   257→    g.addColorStop(0, '#dcfaee')
   258→    g.addColorStop(1, '#7ce3d6')
   259→    ctx.fillStyle = g
   260→    ctx.fillRect(0, 0, 2, 512)
   261→    const tex = new THREE.CanvasTexture(c)
   262→    tex.colorSpace = THREE.SRGBColorSpace
   263→    return tex
   264→  }
   265→
   266→  private makeFloorTexture(): THREE.CanvasTexture {
   267→    const c = document.createElement('canvas')
   268→    c.width = 256
   269→    c.height = 256
   270→    const ctx = c.getContext('2d')!
   271→    ctx.fillStyle = '#b9f1de'
   272→    ctx.fillRect(0, 0, 256, 256)
   273→    ctx.fillStyle = '#a9ecd6'
   274→    for (let y = 0; y < 4; y++)
   275→      for (let x = 0; x < 4; x++) if ((x + y) % 2 === 0) ctx.fillRect(x * 64, y * 64, 64, 64)
   276→    const tex = new THREE.CanvasTexture(c)
   277→    tex.wrapS = THREE.RepeatWrapping
   278→    tex.wrapT = THREE.RepeatWrapping
   279→    tex.repeat.set(10, 10)
   280→    tex.colorSpace = THREE.SRGBColorSpace
   281→    tex.anisotropy = 4
   282→    return tex
   283→  }

   285→  private buildEnvironment() {
   286→    const { trayW: W, trayD: D } = this.def
   287→    const t = 0.5
   288→    const wallH = 1.7
   289→    const frontH = 0.85
   290→
   291→    // outer ground
   292→    const ground = new THREE.Mesh(
   293→      new THREE.PlaneGeometry(60, 60),
   294→      new THREE.MeshStandardMaterial({ map: this.makeFloorTexture(), roughness: 0.85, metalness: 0 })
   295→    )
   296→    ground.rotation.x = -Math.PI / 2
   297→    ground.position.y = -0.01
   298→    ground.receiveShadow = true
   299→    this.scene.add(ground)
   300→
   301→    // tray inner floor
   302→    const tray = new THREE.Mesh(
   303→      new THREE.PlaneGeometry(W, D),
   304→      new THREE.MeshStandardMaterial({ color: 0xd2fbe9, roughness: 0.5, metalness: 0.02 })
   305→    )
   306→    tray.rotation.x = -Math.PI / 2
   307→    tray.position.y = 0.005
   308→    tray.receiveShadow = true
   309→    this.scene.add(tray)
   310→
   311→    // walls
   312→    const wallMat = new THREE.MeshStandardMaterial({ color: 0x7fd8c2, roughness: 0.35, metalness: 0.05 })
   313→    const addWall = (w: number, h: number, d: number, x: number, y: number, z: number) => {
   314→      const mesh = new THREE.Mesh(new RoundedBoxGeometry(w, h, d, 2, 0.1), wallMat)
   315→      mesh.position.set(x, y, z)
   316→      mesh.castShadow = true
   317→      mesh.receiveShadow = true
   318→      this.scene.add(mesh)
   319→      const body = new CANNON.Body({
   320→        mass: 0,
   321→        shape: new CANNON.Box(new CANNON.Vec3(w / 2, h / 2, d / 2)),
   322→        material: this.matGround,
   323→      })
   324→      body.position.set(x, y, z)
   325→      this.world.addBody(body)
   326→    }
   327→    addWall(W + 2 * t, wallH, t, 0, wallH / 2, -D / 2 - t / 2) // back
   328→    addWall(W + 2 * t, frontH, t, 0, frontH / 2, D / 2 + t / 2) // front (low)
   329→    addWall(t, wallH, D, -W / 2 - t / 2, wallH / 2, 0) // left
   330→    addWall(t, wallH, D, W / 2 + t / 2, wallH / 2, 0) // right
   331→  }

   333→  private buildCannon() {
   334→    const { trayD: D } = this.def
   335→    this.cannonPos.set(0, 0, D / 2 + 1.45)
   336→    this.cannonGroup.position.copy(this.cannonPos)
   337→
   338→    const base = new THREE.Mesh(
   339→      new THREE.CylinderGeometry(0.62, 0.72, 0.55, 24),
   340→      new THREE.MeshStandardMaterial({ color: 0x2f6f63, roughness: 0.4, metalness: 0.35 })
   341→    )
   342→    base.position.y = 0.28
   343→    base.castShadow = true
   344→    this.cannonGroup.add(base)
   345→
   346→    const pivot = new THREE.Group()
   347→    pivot.position.y = 0.62
   348→    pivot.rotation.x = -AIM_ELEVATION
   349→    const barrel = new THREE.Mesh(
   350→      new THREE.CylinderGeometry(0.2, 0.27, 1.5, 18),
   351→      new THREE.MeshStandardMaterial({ color: 0x3b8a78, roughness: 0.3, metalness: 0.55 })
   352→    )
   353→    barrel.rotation.x = -Math.PI / 2
   354→    barrel.position.z = -0.72
   355→    barrel.castShadow = true
   356→    pivot.add(barrel)
   357→    this.cannonGroup.add(pivot)
   358→
   359→    // aim arrow (shown when a shot is available)
   360→    const arrowMat = new THREE.MeshBasicMaterial({ color: 0xf59e0b, transparent: true, opacity: 0.55, depthWrite: false })
   361→    const strip = new THREE.Mesh(new THREE.PlaneGeometry(0.32, 2.2), arrowMat)
   362→    strip.rotation.x = -Math.PI / 2
   363→    strip.position.set(0, 0.09, -2.0)
   364→    const head = new THREE.Mesh(new THREE.ConeGeometry(0.3, 0.62, 4), arrowMat)
   365→    head.rotation.x = -Math.PI / 2
   366→    head.position.set(0, 0.09, -3.4)
   367→    this.aimArrow.add(strip, head)
   368→    this.cannonGroup.add(this.aimArrow)
   369→
   370→    this.scene.add(this.cannonGroup)
   371→  }

   373→  private layout() {
   374→    const w = this.canvas.clientWidth || window.innerWidth
   375→    const h = this.canvas.clientHeight || window.innerHeight
   376→    this.renderer.setSize(w, h, false)
   377→    const aspect = w / h
   378→    this.camera.aspect = aspect
   379→    const vFov = THREE.MathUtils.degToRad(this.camera.fov)
   380→    const hFov = 2 * Math.atan(Math.tan(vFov / 2) * aspect)
   381→    const distW = (this.def.trayW / 2 + 1.4) / Math.tan(hFov / 2)
   382→    const distD = ((this.def.trayD / 2 + 2.6) / Math.tan(vFov / 2)) * 0.74
   383→    const dist = Math.max(distW, distD, Math.max(this.def.trayW, this.def.trayD) * 1.2)
   384→    this.camBase.set(0, dist * 0.78, dist * 0.64)
   385→    this.camLook.set(0, 0, this.def.trayD * 0.06)
   386→    this.camera.updateProjectionMatrix()
   387→  }

   391→  private tierGeometry(tier: number): THREE.BufferGeometry {
   392→    if (!this.tierGeos[tier]) {
   393→      const t = TIERS[tier]
   394→      switch (t.shape) {
   395→        case 'sphere':
   396→          this.tierGeos[tier] = new THREE.SphereGeometry(t.r ?? 0.4, 22, 14)
   397→          break
   398→        case 'box':
   399→          this.tierGeos[tier] = new RoundedBoxGeometry(t.s ?? 0.8, t.s ?? 0.8, t.s ?? 0.8, 3, 0.11)
   400→          break
   401→        case 'cylinder':
   402→          this.tierGeos[tier] = new THREE.CylinderGeometry(t.r ?? 0.5, t.r ?? 0.5, t.h ?? 0.9, 22)
   403→          break
   404→        case 'ico':
   405→          this.tierGeos[tier] = new THREE.IcosahedronGeometry(t.r ?? 0.7, 0)
   406→          break
   407→        case 'gem':
   408→          this.tierGeos[tier] = new RoundedBoxGeometry(t.s ?? 1.2, (t.s ?? 1.2) * 0.74, t.s ?? 1.2, 3, 0.16)
   409→          break
   410→      }
   411→    }
   412→    return this.tierGeos[tier]
   413→  }

   420→  private makeItem(tier: number, x: number, y: number, z: number): MergeItem {
   421→    const def = TIERS[tier]
   422→    const shape = this.tierShapeFor(tier)
   423→    const body = new CANNON.Body({
   424→      mass: def.mass,
   425→      shape,
   426→      position: new CANNON.Vec3(x, y, z),
   427→      material: this.matObj,
   428→      angularVelocity: new CANNON.Vec3(0, (Math.random() - 0.5) * 1.5, 0),
   429→    })
   430→    body.linearDamping = 0.08
   431→    body.angularDamping = 0.25
   432→    body.allowSleep = true
   433→    body.sleepSpeedLimit = 0.25
   434→    body.sleepTimeLimit = 0.6
   435→    this.world.addBody(body)
   436→
   437→    const mat = new THREE.MeshStandardMaterial({
   438→      color: def.color,
   439→      roughness: 0.16,
   440→      metalness: 0.08,
   441→      emissive: def.color,
   442→      emissiveIntensity: def.glow ?? 0.07,
   443→      flatShading: def.shape === 'ico',
   444→    })
   445→    const mesh = new THREE.Mesh(this.tierGeometry(tier), mat)
   446→    mesh.castShadow = true
   447→    mesh.receiveShadow = true
   448→    this.scene.add(mesh)
   449→
   450→    const item: MergeItem = { body, mesh, tier, born: this.time, held: false, consumed: false }
   451→    mesh.userData.item = item
   452→    this.items.set(body.id, item)
   453→    body.addEventListener('collide', this.onItemCollide)
   454→    return item
   455→  }

   457→  private tierShapeFor(tier: number): CANNON.Shape {
   458→    const t = TIERS[tier]
   459→    switch (t.shape) {
   460→      case 'sphere':
   461→        return new CANNON.Sphere(t.r ?? 0.4)
   462→      case 'box':
   463→        return new CANNON.Box(new CANNON.Vec3((t.s ?? 0.8) / 2, (t.s ?? 0.8) / 2, (t.s ?? 0.8) / 2))
   464→      case 'cylinder':
   465→        return new CANNON.Cylinder(t.r ?? 0.5, t.r ?? 0.5, t.h ?? 0.9, 12)
   466→      case 'ico':
   467→        return new CANNON.Sphere(t.r ?? 0.7)
   468→      case 'gem':
   469→        return new CANNON.Box(new CANNON.Vec3((t.s ?? 1.2) / 2, ((t.s ?? 1.2) * 0.74) / 2, (t.s ?? 1.2) / 2))
   470→    }
   471→  }

   473→  private removeItem(item: MergeItem) {
   474→    if (item.consumed) return
   475→    item.consumed = true
   476→    item.body.removeEventListener('collide', this.onItemCollide)
   477→    this.world.removeBody(item.body)
   478→    this.scene.remove(item.mesh)
   479→    ;(item.mesh.material as THREE.Material).dispose()
   480→    this.items.delete(item.body.id)
   481→    if (this.held === item) this.held = null
   482→  }

   484→  private makeObstacle(o: LevelDef['obstacles'][number]) {
   485→    const style = OBSTACLE_STYLE[o.kind]
   486→    const mass = o.kind === 'crate' ? 6 : o.kind === 'crystal' ? 10 : 14
   487→    const body = new CANNON.Body({
   488→      mass,
   489→      shape: new CANNON.Box(new CANNON.Vec3(o.w / 2, o.h / 2, o.d / 2)),
   490→      position: new CANNON.Vec3(o.x, o.h / 2 + 0.05, o.z),
   491→      material: this.matObj,
   492→    })
   493→    if (o.rot) body.quaternion.setFromEuler(0, o.rot, 0)
   494→    body.linearDamping = 0.4
   495→    body.angularDamping = 0.6
   496→    body.allowSleep = true
   497→    body.sleepSpeedLimit = 0.3
   498→    body.sleepTimeLimit = 0.6
   499→    this.world.addBody(body)
   500→
   501→    const mat = new THREE.MeshStandardMaterial({
   502→      color: style.color,
   503→      roughness: o.kind === 'crystal' ? 0.15 : 0.62,
   504→      metalness: o.kind === 'crystal' ? 0.35 : 0.05,
   505→      transparent: o.kind === 'crystal',
   506→      opacity: o.kind === 'crystal' ? 0.88 : 1,
   507→      emissive: 0x000000,
   508→    })
   509→    const mesh = new THREE.Mesh(this.obstacleGeo, mat)
   510→    mesh.scale.set(o.w, o.h, o.d)
   511→    mesh.castShadow = true
   512→    mesh.receiveShadow = true
   513→    this.scene.add(mesh)
   514→
   515→    this.obstacles.push({ body, mesh, hp: o.hp, maxHp: o.hp, kind: o.kind, flash: 0, dead: false })
   516→  }

   534→  private onPointerDown = (e: PointerEvent) => {

   565→    }

   626→  private fire() {
   627→    if (this.ball || this.shots <= 0 || this.won || this.lost) return
   628→    this.shots--
   629→    this.lastShotTime = this.time
   630→    this.cb.onShots(this.shots)
   631→
   632→    let isBomb = false
   633→    if (this.bombArmed) {
   634→      isBomb = true
   635→      this.bombArmed = false
   636→      this.cb.onBombArmed(false)
   637→      this.cb.onBombConsumed()
   638→    }
   639→
   640→    audio.shoot()
   641→    this.buzz(25)
   642→    this.shake = Math.max(this.shake, 0.14)
   643→
   644→    const dir = new THREE.Vector3(Math.sin(this.aimAngle), 0, -Math.cos(this.aimAngle))
   645→    const spawn = new THREE.Vector3(
   646→      this.cannonPos.x + dir.x * 0.85,
   647→      1.05,
   648→      this.cannonPos.z + dir.z * 0.85
   649→    )
   650→    const body = new CANNON.Body({
   651→      mass: 3.2,
   652→      shape: new CANNON.Sphere(0.26),
   653→      position: new CANNON.Vec3(spawn.x, spawn.y, spawn.z),
   654→      material: this.matBall,
   655→    })
   656→    body.linearDamping = 0.05
   657→    body.angularDamping = 0.1
   658→    body.velocity.set(dir.x * (16.5 + Math.abs(this.aimAngle) * 2), 6.0, dir.z * (16.5 + Math.abs(this.aimAngle) * 2))
   659→    this.world.addBody(body)
   660→
   661→    const skin = BALL_SKINS.find((s) => s.id === this.skinId) ?? BALL_SKINS[0]
   662→    const mat = new THREE.MeshStandardMaterial({
   663→      color: skin.color,
   664→      roughness: skin.id === 'gold' ? 0.18 : 0.15,
   665→      metalness: skin.id === 'gold' ? 0.95 : skin.id === 'neon' ? 0.1 : 0.45,
   666→      emissive: skin.id === 'neon' ? new THREE.Color(skin.color) : new THREE.Color(0x000000),
   667→      emissiveIntensity: skin.id === 'neon' ? 0.9 : 0,
   668→    })
   669→    const mesh = new THREE.Mesh(new THREE.SphereGeometry(0.26, 20, 14), mat)
   670→    mesh.castShadow = true
   671→    this.scene.add(mesh)
   672→
   673→    const ballItem: BallItem = {
   674→      body,
   675→      mesh,
   676→      born: this.time,
   677→      slowTime: 0,
   678→      isBomb,
   679→      onCollide: () => {},
   680→    }
   681→    ballItem.onCollide = (evt: CollideEvt) => {
   682→      const v = Math.abs(evt.contact.getImpactVelocityAlongNormal())
   683→      if (v < 1.2) return
   684→      const other = evt.body
   685→      const obstacle = this.obstacles.find((o) => o.body.id === other.id && !o.dead)
   686→      if (obstacle) {
   687→        const dmg = (v - 1.5) * (isBomb ? 5.5 : 3.0)
   688→        this.pendingHits.push({ obstacle, dmg: Math.max(4, dmg), point: body.position.clone(), bomb: false })
   689→      } else if (v > 4.5) {
   690→        audio.thud()
   691→      }
   692→      if (isBomb && v > 2) {
   693→        this.pendingHits.push({ obstacle: null, dmg: 0, point: body.position.clone(), bomb: true })
   694→      }
   695→    }
   696→    body.addEventListener('collide', ballItem.onCollide)
   697→    this.ball = ballItem
   698→  }

   716→    if (this.pendingHits.length === 0) return

   718→    this.pendingHits = []

   720→    for (const h of hits) {

   722→        if (bombDone) continue
   723→        bombDone = true

   733→        continue
   734→      }
   735→      if (h.obstacle && !h.obstacle.dead) this.damageObstacle(h.obstacle, h.dmg, h.point)

   773→  private splashDamage(center: CANNON.Vec3, radius: number, dmg: number) {
   774→    for (const o of [...this.obstacles]) {
   775→      if (o.dead) continue
   776→      const d = o.body.position.distanceTo(center)
   777→      if (d < radius) this.damageObstacle(o, dmg * (1 - d / radius) + 4, center)
   778→    }

   781→  private explode(center: CANNON.Vec3, radius: number, strength: number) {
   782→    for (const item of this.items.values()) {
   783→      const d = item.body.position.distanceTo(center)
   784→      if (d > radius || item.held) continue
   785→      item.body.wakeUp()
   786→      const dir = item.body.position.vsub(center)
   787→      dir.y = Math.abs(dir.y) + 1.2
   788→      dir.normalize()
   789→      const power = strength * (1 - d / radius) * (TIERS[item.tier].mass * 0.6 + 1)
   790→      item.body.applyImpulse(dir.scale(power))
   791→    }
   792→  }

   796→  private queuePair(evt: CollideEvt) {
   797→    const a = this.items.get(evt.target.id)
   798→    const b = this.items.get(evt.body.id)
   799→    if (!a || !b || a.consumed || b.consumed) return
   800→    if (a.tier !== b.tier) return
   801→    const key = a.body.id < b.body.id ? `${a.body.id}_${b.body.id}` : `${b.body.id}_${a.body.id}`
   802→    if (!this.pendingPairs.has(key)) this.pendingPairs.set(key, [a, b])
   803→  }

   805→  private processMerges() {
   806→    if (this.pendingPairs.size === 0) return
   807→    const pairs = [...this.pendingPairs.values()]
   808→    this.pendingPairs.clear()
   809→    for (const [a, b] of pairs) {
   810→      if (a.consumed || b.consumed || a.tier !== b.tier) continue
   811→      if (!this.items.has(a.body.id) || !this.items.has(b.body.id)) continue
   812→      this.doMerge(a, b)
   813→    }
   814→  }

   816→  private doMerge(a: MergeItem, b: MergeItem) {
   817→    const mid = new THREE.Vector3(
   818→      (a.body.position.x + b.body.position.x) / 2,
   819→      Math.max(a.body.position.y, b.body.position.y),
   820→      (a.body.position.z + b.body.position.z) / 2
   821→    )
   822→    const vx = (a.body.velocity.x + b.body.velocity.x) / 2
   823→    const vz = (a.body.velocity.z + b.body.velocity.z) / 2
   824→    const tier = a.tier
   825→    const color = TIERS[tier].color
   826→    this.removeItem(a)
   827→    this.removeItem(b)
   828→
   829→    // ---- STAR BLAST: two max-tier gems explode
   830→    if (tier >= MAX_TIER) {
   831→      this.addScore(500, mid.clone().setY(1.6))
   832→      this.fx.floatText('انفجار ستاره‌ای!', mid.clone().setY(2.2), 'big')
   833→      this.fx.burst(mid, '#f472b6', 60, 8, 4)
   834→      this.fx.burst(mid, '#ffffff', 30, 6, 3)
   835→      this.fx.ring(mid, '#f472b6', 4.2)
   836→      audio.blast()
   837→      this.buzz(90)
   838→      this.shake = Math.max(this.shake, 0.8)
   839→      const c = new CANNON.Vec3(mid.x, mid.y, mid.z)
   840→      this.explode(c, 3.8, 10)
   841→      this.splashDamage(c, 3.2, 50)
   842→      return
   843→    }
   844→
   845→    const newTier = tier + 1
   846→    const item = this.makeItem(newTier, mid.x, Math.max(0.9, mid.y), mid.z)
   847→    item.body.velocity.set(vx, 3.2, vz)
   848→    item.born = this.time
   849→
   850→    // combo & score
   851→    if (this.time < this.comboUntil) this.combo++
   852→    else this.combo = 1
   853→    this.comboUntil = this.time + 1.35
   854→    const mult = Math.min(1 + (this.combo - 1) * 0.25, 3)
   855→    const gained = Math.round(TIERS[newTier].score * mult)
   856→    this.addScore(gained, mid.clone().setY(mid.y + 1.1))
   857→    this.cb.onCombo(this.combo)
   858→
   859→    // fx
   860→    this.fx.burst(mid, color, 22 + newTier * 6, 4.5 + newTier, 2.4)
   861→    this.fx.ring(mid, color, 1.6 + newTier * 0.5)
   862→    audio.merge(newTier)
   863→    this.buzz(18 + newTier * 12)
   864→    if (newTier === MAX_TIER) {
   865→      this.fx.floatText('ستاره‌ی درخشان!', mid.clone().setY(2.1), 'big')
   866→      this.fx.burst(mid, '#ffffff', 24, 6, 3)
   867→      this.shake = Math.max(this.shake, 0.3)
   868→    }
   869→
   870→    // goal
   871→    if (this.def.goal.type === 'merge' && newTier === this.def.goal.tier) {
   872→      this.goalCurrent++
   873→      this.cb.onGoal(this.goalCurrent, this.def.goal.count ?? 1)
   874→      this.checkWin()
   875→    }
   876→  }

   879→    if (n === 0) return
   880→    this.score += n

   890→  // ======================================================= win / lose

   892→  private checkWin() {
   893→    if (this.won || this.lost) return
   894→    const g = this.def.goal
   895→    let done = false
   896→    if (g.type === 'merge') done = this.goalCurrent >= (g.count ?? 1)
   897→    else if (g.type === 'clear') done = this.obstacles.length === 0
   898→    else done = this.score >= (g.points ?? Infinity)
   899→    if (!done) return
   900→    this.won = true
   901→    this.winAt = this.time
   902→    audio.win()
   903→    this.buzz([40, 60, 40, 60, 80])
   904→    // confetti
   905→    for (let i = 0; i < 5; i++) {
   906→      const p = new THREE.Vector3((Math.random() - 0.5) * this.def.trayW * 0.8, 1.5, (Math.random() - 0.5) * this.def.trayD * 0.7)
   907→      this.fx.burst(p, ['#f472b6', '#fbbf24', '#5eead4', '#b18cfe', '#ff8f6b'][i], 24, 6, 4)
   908→    }
   909→  }
   910→
   911→  private checkFail() {
   912→    if (this.won || this.lost || this.time < 2.5) return
   913→    if (this.def.goal.type === 'clear') {
   914→      if (
   915→        this.shots === 0 &&
   916→        !this.ball &&
   917→        !this.bombArmed &&
   918→        this.obstacles.length > 0 &&
   919→        this.time - this.lastBreakTime > 1.6 &&
   920→        this.time - this.lastShotTime > 1.4
   921→      ) {
   922→        this.lose('توپ‌ها تمام شد ولی موانع باقی مانده‌اند!')
   923→      }
   924→      return
   925→    }
   926→    // stuck detection (no possible merge left)
   927→    let pairs = false
   928→    const counts = new Map<number, number>()
   929→    for (const it of this.items.values()) counts.set(it.tier, (counts.get(it.tier) ?? 0) + 1)
   930→    for (const c of counts.values()) if (c >= 2) pairs = true
   931→    if (pairs || this.held || this.pendingPairs.size > 0) {
   932→      this.stuckSince = 0
   933→      return
   934→    }
   935→    // for score goals, breaking obstacles still earns points
   936→    if (this.def.goal.type === 'score' && this.obstacles.length > 0 && (this.shots > 0 || this.ball || this.bombArmed)) return
   937→    if (this.stuckSince === 0) this.stuckSince = this.time
   938→    else if (this.time - this.stuckSince > 2.2) this.lose('حرکت ممکنی نماند!')
   939→  }

   941→  private lose(reason: string) {
   942→    if (this.won || this.lost) return
   943→    this.lost = true
   944→    this.loseReason = reason
   945→    this.lostAt = this.time
   946→    if (this.held) this.releaseHeld()
   947→    audio.lose()
   948→    this.buzz([60, 40, 60])
   949→  }

   951→  // ================================================================ loop
   952→
   953→  private loop = () => {
   954→    if (this.disposed) return
   955→    this.rafId = requestAnimationFrame(this.loop)
   956→    const dt = Math.min(this.clock.getDelta(), 0.05)
   957→    if (this.paused) {
   958→      this.render()
   959→      return
   960→    }
   961→    this.time += dt
   962→
   963→    // physics
   964→    this.world.step(1 / 60, dt, 3)
   965→    this.processHits()
   966→    this.processMerges()
   967→
   968→    // held object follows finger (kinematic velocity steering)
   969→    if (this.held) {
   970→      const b = this.held.body
   971→      const dx = this.dragPoint.x - b.position.x
   972→      const dz = this.dragPoint.z - b.position.z
   973→      const vx = THREE.MathUtils.clamp(dx * 14, -22, 22)
   974→      const vz = THREE.MathUtils.clamp(dz * 14, -22, 22)
   975→      b.velocity.set(vx, 0, vz)
   976→      b.wakeUp()
   977→    }
   978→
   979→    // sync meshes
   980→    for (const item of this.items.values()) {
   981→      item.mesh.position.set(item.body.position.x, item.body.position.y, item.body.position.z)
   982→      item.mesh.quaternion.set(item.body.quaternion.x, item.body.quaternion.y, item.body.quaternion.z, item.body.quaternion.w)
   983→      const t = (this.time - item.born) / 0.28
   984→      item.mesh.scale.setScalar(t < 1 ? Math.max(0.12, easeOutBack(Math.max(0, t))) : 1)
   985→      if (item.tier === MAX_TIER) {
   986→        ;(item.mesh.material as THREE.MeshStandardMaterial).emissiveIntensity =
   987→          (TIERS[MAX_TIER].glow ?? 0.5) * (0.75 + 0.25 * Math.sin(this.time * 4))
   988→      }
   989→    }
   990→    for (const o of this.obstacles) {
   991→      o.mesh.position.set(o.body.position.x, o.body.position.y, o.body.position.z)
   992→      o.mesh.quaternion.set(o.body.quaternion.x, o.body.quaternion.y, o.body.quaternion.z, o.body.quaternion.w)
   993→      if (o.flash > 0) {
   994→        o.flash = Math.max(0, o.flash - dt * 3)
   995→        const ratio = 1 - o.hp / o.maxHp
   996→        ;(o.mesh.material as THREE.MeshStandardMaterial).emissive.setRGB(0.9 * o.flash + ratio * 0.25, 0.15 * o.flash, 0.05 * o.flash)
   997→      }
   998→    }
   999→    if (this.ball) {
  1000→      this.ball.mesh.position.set(this.ball.body.position.x, this.ball.body.position.y, this.ball.body.position.z)
  1001→      this.ball.mesh.quaternion.set(
  1002→        this.ball.body.quaternion.x,
  1003→        this.ball.body.quaternion.y,
  1004→        this.ball.body.quaternion.z,
  1005→        this.ball.body.quaternion.w
  1006→      )
  1007→      const speed = this.ball.body.velocity.length()
  1008→      if (speed < 0.8) this.ball.slowTime += dt
  1009→      else this.ball.slowTime = 0
  1010→      if (this.time - this.ball.born > 8 || this.ball.slowTime > 1.4) this.removeBall(true)
  1011→    }
  1012→
  1013→    // combo expiry
  1014→    if (this.combo > 0 && this.time > this.comboUntil) {
  1015→      this.combo = 0
  1016→      this.cb.onCombo(0)
  1017→    }
  1018→
  1019→    // aim arrow visibility
  1020→    this.canShoot = !this.ball && this.shots > 0 && !this.won && !this.lost
  1021→    this.aimArrow.visible = this.canShoot
  1022→    if (this.canShoot) {
  1023→      const child = this.aimArrow.children[0] as THREE.Mesh
  1024→      ;(child.material as THREE.MeshBasicMaterial).opacity = this.aiming ? 0.9 : 0.5 + 0.12 * Math.sin(this.time * 5)
  1025→    }
  1026→
  1027→    // win / lose deferred callbacks
  1028→    if (this.won && !this.winFired && this.time - this.winAt > 1.15) {
  1029→      this.winFired = true
  1030→      this.cb.onWin(this.score)
  1031→    }
  1032→    if (this.lost && !this.loseFired && this.time - this.lostAt > 0.95) {
  1033→      this.loseFired = true
  1034→      this.cb.onLose(this.loseReason)
  1035→    }
  1036→
  1037→    // periodic fail check
  1038→    this.failTimer += dt
  1039→    if (this.failTimer > 0.5) {
  1040→      this.failTimer = 0
  1041→      this.checkFail()
  1042→    }
  1043→
  1044→    this.fx.update(dt)
  1045→    this.render()
  1046→  }
