    97→    97→function easeOutBack(t: number): number {
    98→    98→  const c1 = 1.70158
    99→    99→  const c3 = c1 + 1
   100→   100→  return 1 + c3 * Math.pow(t - 1, 3) + c1 * Math.pow(t - 1, 2)

   103→   103→function roundedRectPath(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   104→   104→  const rad = Math.min(r, h / 2, w / 2)
   105→   105→  ctx.beginPath()
   106→   106→  ctx.moveTo(x + rad, y)
   107→   107→  ctx.arcTo(x + w, y, x + w, y + h, rad)
   108→   108→  ctx.arcTo(x + w, y + h, x, y + h, rad)
   109→   109→  ctx.arcTo(x, y + h, x, y, rad)
   110→   110→  ctx.arcTo(x, y, x + w, y, rad)
   111→   111→  ctx.closePath()
   112→   112→}

   348→   348→  private makeWoodTexture(): THREE.CanvasTexture {
   349→   349→    const c = document.createElement('canvas')
   350→   350→    c.width = 128
   351→   351→    c.height = 128
   352→   352→    const ctx = c.getContext('2d')!
   353→   353→    const shades = ['#cf9058', '#c68a52', '#d29a63', '#c2854f']
   354→   354→    for (let i = 0; i < 4; i++) {
   355→   355→      ctx.fillStyle = shades[i]
   356→   356→      ctx.fillRect(0, i * 32, 128, 32)
   357→   357→      // grain streaks
   358→   358→      ctx.strokeStyle = 'rgba(122,74,38,0.28)'
   359→   359→      ctx.lineWidth = 1.4
   360→   360→      for (let g = 0; g < 3; g++) {
   361→   361→        const gy = i * 32 + 6 + g * 9 + Math.random() * 4
   362→   362→        ctx.beginPath()
   363→   363→        ctx.moveTo(4, gy)
   364→   364→        ctx.bezierCurveTo(40, gy - 3, 80, gy + 3, 124, gy)
   365→   365→        ctx.stroke()
   366→   366→      }
   367→   367→      // plank separator
   368→   368→      ctx.fillStyle = 'rgba(96,58,28,0.75)'
   369→   369→      ctx.fillRect(0, i * 32, 128, 2.4)
   370→   370→    }
   371→   371→    // nails
   372→   372→    ctx.fillStyle = '#7a4a26'
   373→   373→    for (const [nx, ny] of [[8, 6], [120, 6], [8, 122], [120, 122]]) {
   374→   374→      ctx.beginPath()
   375→   375→      ctx.arc(nx, ny, 2.6, 0, Math.PI * 2)
   376→   376→      ctx.fill()
   377→   377→    }
   378→   378→    const tex = new THREE.CanvasTexture(c)
   379→   379→    tex.colorSpace = THREE.SRGBColorSpace
   380→   380→    tex.anisotropy = 4

   773→   773→  private onPointerDown = (e: PointerEvent) => {
   774→   774→    audio.unlock()
   775→   775→    if (this.paused || this.won || this.lost || this.disposed) return
   776→   776→    this.setNdc(e)
   777→   777→
   778→   778→    const rect = this.canvas.getBoundingClientRect()
   779→   779→    const inStrip = e.clientY - rect.top > rect.height * 0.83
   780→   780→
   781→   781→    // 1) bottom strip ALWAYS starts aiming (never steals an item drag)
   782→   782→    if (inStrip) {
   783→   783→      if (!this.canShoot) return
   784→   784→      this.aiming = true
   785→   785→      this.updateAim()
   786→   786→      this.canvas.setPointerCapture(e.pointerId)
   787→   787→      return
   788→   788→    }
   789→   789→
   790→   790→    // 2) pick a mergeable object
   791→   791→    this.raycaster.setFromCamera(this.ndc, this.camera)
   792→   792→    const meshes: THREE.Mesh[] = []
   793→   793→    for (const it of this.items.values()) if (!it.consumed) meshes.push(it.mesh)
   794→   794→    const hits = this.raycaster.intersectObjects(meshes, false)
   795→   795→    if (hits.length > 0) {
   796→   796→      const item = hits[0].object.userData.item as MergeItem
   797→   797→      this.held = item
   798→   798→      item.held = true
   799→   799→      item.body.type = CANNON.Body.KINEMATIC
   800→   800→      item.body.velocity.setZero()
   801→   801→      item.body.angularVelocity.scale(0.2, item.body.angularVelocity)
   802→   802→      item.body.wakeUp()
   803→   803→      this.holdPlane.set(new THREE.Vector3(0, 1, 0), -HOLD_Y)
   804→   804→      this.updateDragPoint()
   805→   805→      audio.pick()
   806→   806→      this.canvas.setPointerCapture(e.pointerId)
   807→   807→    }
   808→   808→  }

   835→   835→  private releaseHeld() {
   836→   836→    const item = this.held

   847→   847→  private updateDragPoint() {
   848→   848→    if (!this.held) return
   849→   849→    this.raycaster.setFromCamera(this.ndc, this.camera)
   850→   850→    const hit = new THREE.Vector3()
   851→   851→    if (!this.raycaster.ray.intersectPlane(this.holdPlane, hit)) return
   852→   852→    const r = this.itemRadius(this.held.tier)
   853→   853→    const hw = this.def.trayW / 2 - r - 0.08
   854→   854→    const hd = this.def.trayD / 2 - r - 0.08
   855→   855→    hit.x = THREE.MathUtils.clamp(hit.x, -hw, hw)
   856→   856→    hit.z = THREE.MathUtils.clamp(hit.z, -hd, hd)
   857→   857→    hit.y = HOLD_Y
   858→   858→    this.dragPoint.copy(hit)
   859→   859→  }

   863→   863→   * slides toward it and aims at it. Nearby obstacles magnetically snap
   864→   864→   * the target onto their center (aim assist). The target is clamped to
   865→   865→   * the tray interior so the aim direction stays stable.

   869→   869→    const hit = new THREE.Vector3()

   871→   871→

   874→   874→    // keep the target strictly in front of the cannon (inside the tray)
   875→   875→    hit.z = THREE.MathUtils.clamp(hit.z, -this.def.trayD / 2, this.def.trayD / 2 - 0.3)
   876→   876→    hit.y = AIM_PLANE_Y
   877→   877→

   879→   879→    this.aimTarget.copy(hit)

   881→   881→    let snapped: ObstacleItem | null = null
   882→   882→    for (const o of this.obstacles) {
   883→   883→      const d = Math.hypot(o.body.position.x - hit.x, o.body.position.z - hit.z)
   884→   884→      if (d < best) {
   885→   885→        best = d
   886→   886→        snapped = o
   887→   887→      }
   888→   888→    }
   889→   889→    if (snapped) this.aimTarget.set(snapped.body.position.x, AIM_PLANE_Y, snapped.body.position.z)
   890→   890→

   892→   892→  }

  1406→  1406→  /**
  1407→  1407→   * Simulates the exact shot physics for a dotted preview and shows a

  1410→  1410→  private updateTrajectory() {
  1411→  1411→    const dir = new THREE.Vector3(Math.sin(this.aimAngle), 0, -Math.cos(this.aimAngle))
  1412→  1412→    const speed = SHOT_SPEED + Math.abs(this.aimAngle) * 1.5
  1413→  1413→    const pos = new THREE.Vector3(
  1414→  1414→      this.cannonGroup.position.x + dir.x * 0.95,
  1415→  1415→      1.05,
  1416→  1416→      this.cannonPos.z + dir.z * 0.95
  1417→  1417→    )
  1418→  1418→    const vel = new THREE.Vector3(dir.x * speed, SHOT_VY, dir.z * speed)
  1419→  1419→    const step = 1 / 40
  1420→  1420→    const damp = Math.pow(1 - 0.05, step)
  1421→  1421→    let impact: ObstacleItem | null = null
  1422→  1422→    let dotIdx = 0
  1423→  1423→
  1424→  1424→    for (let t = 0; t < 2.4 && dotIdx < TRAJ_DOTS; t += step) {
  1425→  1425→      vel.y -= GRAVITY * step
  1426→  1426→      vel.multiplyScalar(damp)
  1427→  1427→      pos.addScaledVector(vel, step)
  1428→  1428→
  1429→  1429→      // obstacle intersection (expanded AABB)
  1430→  1430→      for (const o of this.obstacles) {
  1431→  1431→        const hx = o.w / 2 + 0.15
  1432→  1432→        const hy = o.h / 2 + 0.15
  1433→  1433→        const hz = o.d / 2 + 0.15
  1434→  1434→        if (
  1435→  1435→          Math.abs(pos.x - o.body.position.x) < hx &&
  1436→  1436→          Math.abs(pos.y - o.body.position.y) < hy &&
  1437→  1437→          Math.abs(pos.z - o.body.position.z) < hz
  1438→  1438→        ) {
  1439→  1439→          impact = o
  1440→  1440→          break
  1441→  1441→        }
  1442→  1442→      }
  1443→  1443→      const outOfTray =
  1444→  1444→        pos.y < 0.25 || Math.abs(pos.x) > this.def.trayW / 2 + 0.2 || pos.z < -this.def.trayD / 2 - 0.2
  1445→  1445→      if (impact || outOfTray) break
  1446→  1446→
  1447→  1447→      const dot = this.trajDots[dotIdx]
  1448→  1448→      dot.visible = true
  1449→  1449→      dot.position.copy(pos)
  1450→  1450→      dot.scale.setScalar(1 - (dotIdx / TRAJ_DOTS) * 0.55)
  1451→  1451→      dotIdx++
  1452→  1452→    }
  1453→  1453→    for (let i = dotIdx; i < TRAJ_DOTS; i++) this.trajDots[i].visible = false
  1454→  1454→    this.trajGroup.visible = true
  1455→  1455→    this.trajMat.opacity = this.aiming ? 0.9 : 0.35
  1456→  1456→
  1457→  1457→    // reticle on the predicted impact
  1458→  1458→    const rx = impact ? impact.body.position.x : THREE.MathUtils.clamp(pos.x, -this.def.trayW / 2, this.def.trayW / 2)
  1459→  1459→    const rz = impact ? impact.body.position.z : THREE.MathUtils.clamp(pos.z, -this.def.trayD / 2, this.def.trayD / 2)
  1460→  1460→    this.reticle.visible = true
  1461→  1461→    this.reticle.position.set(rx, 0, rz)
  1462→  1462→    const pulse = 1 + 0.12 * Math.sin(this.time * 6)
  1463→  1463→    this.reticle.scale.setScalar(pulse)
  1464→  1464→    this.reticle.rotation.y += 0.02
  1465→  1465→    this.reticleMat.color.set(impact ? 0xf59e0b : 0xffffff)
  1466→  1466→    this.reticleMat.opacity = (this.aiming ? 0.95 : 0.5) * pulse
  1467→  1467→  }
